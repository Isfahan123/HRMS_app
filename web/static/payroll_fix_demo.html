window.downloadPayslip = async function(employeeId, payrollRunId) { // Fetch PDF from API const response = await fetch(`/api/payroll/payslip/${employeeId}/${payrollRunId}`); const blob = await response.blob(); // Trigger browser download const url = window.URL.createObjectURL(blob); const a = document.createElement('a'); a.href = url; a.download = `payslip_${employeeId}_${payrollRunId}.pdf`; a.click(); };
