# HRMS Web Application - Production Dependencies
# For web deployment on cPanel, Exabytes, or similar hosting
#
# PYTHON VERSION REQUIREMENT: Python >= 3.9 (Python 3.11 recommended)
#
# NOTE: This file is for web hosting ONLY.
# For desktop GUI application, also install requirements-desktop.txt:
#   pip install -r requirements.txt -r requirements-desktop.txt
#
# INSTALLATION TROUBLESHOOTING:
# If pip install fails on cPanel, use the minimal requirements file:
#   pip install -r requirements-minimal.txt
# The minimal version excludes PDF generation (fpdf2/Pillow) which may
# fail to compile on some hosting environments.
#
# IMPORTANT: All core packages here are pure Python or have pre-built wheels
# to ensure compatibility with cPanel/shared hosting environments.

# Database
# NOTE: We use a custom REST API client (services/supabase_rest_client.py) instead of
# the official supabase-py SDK because the SDK requires Rust compilation which fails
# on cPanel/shared hosting. The REST client uses the 'requests' library directly.
# No supabase package needed - using requests for REST API calls

# Authentication & Security
# Using passlib with pbkdf2_sha256 (pure Python) instead of bcrypt (requires C compilation)
passlib==1.7.4

# Utilities (all pure Python)
pytz==2024.1
python-dotenv==1.0.1
requests==2.32.3
num2words==0.5.13
# Note: 'holidays' library removed - uses compiled numpy/pandas dependencies
# Holiday data is now fetched from database (calendar_holidays table) or Calendarific API

# Data processing (required for payroll reports)
# fpdf2 requires Pillow and fonttools which may need compilation on some systems.
# If this fails, use requirements-minimal.txt instead - PDF generation will be disabled.
# Use fpdf2 2.7.6 for Python 3.7-3.9 compatibility, or 2.8.1+ for Python 3.10+
fpdf2==2.7.6
# Note: pandas removed - requires numpy which needs C compilation
# Excel parsing now uses openpyxl directly (pure Python)
openpyxl==3.1.2
PyPDF2==3.0.1

# Flask stack (Web Framework)
# Flask is used instead of FastAPI because:
# - cPanel Passenger only supports WSGI (not ASGI)
# - Flask is native WSGI - no conversion layer needed
# - More reliable on shared hosting environments
# - No uvicorn/a2wsgi dependencies that can fail to install
flask==3.0.0
jinja2==3.1.4
